    \item Suponga el siguiente problema de maximización:
\end{enumerate}

\begin{align}
    \max_{c_1,c_2}&= ln C_1 + \beta lnC_2\\  
    s.a: C_1&= Y_1 + \frac{B_2}{R}\\
         C_2&= Y_2 - B_2
\end{align}

\begin{enumerate}[label=\alph*)]
    \item Plantear el Lagrangeano.
    \item Encontrar Euler.
    \item Analice que ocurre con el consumo dado: $\beta$R $<$ 1.
    \item Analice que ocurre con el consumo dado: $\beta$R = 1.
    \item Analice que ocurre con el consumo dado: $\beta$R $>$ 1.
    \item Repita a) y b) pero ahora con el siguiente problema de maximización.


\begin{align}
   U&=\frac{C^{1-\sigma}-1}{1-\sigma} \\  
    s.a&: C_1 + \frac{C_2}{1+R}=B
\end{align}

\item Explique que significa $\sigma$.

\end{enumerate}
