\section*{Ejercicio}

\begin{enumerate}  \item las familias desean maximizar su utilidad, donde el problema de maximización queda expresado de la siguiente manera:  
\end{enumerate}
\begin{equation*}
\max U(C_t,B_t, M_t): ln(C_1)+ \beta ln(C_2) + ln \left(\frac{M_1}{P_1}\right)+\beta ln \left(\frac{M_2}{P_2}\right)
\end{equation*}
\begin{align*}
    s.a \hspace{1cm} P_1 C_1=&P_1 Y + \frac{B_2}{R}- M_1 - T_1\\
     P_2 C_2=&P_2 Y - B_2 + M_1 - M_2 - T_2
\end{align*}

\begin{enumerate}[label=\alph*)]
\item Transformar las variables a términos reales.
\item Plantear el problema de máximización. 
\item Encontrar Euler- Lagrange de consumo.
\item Resolver los siguientes casos
\begin{enumerate}[label=\alph*.\arabic*)]
        \item Se desea que $C_2=C_1$.
        \item No se interviene .
        \end{enumerate}
\item Resuelva ahora suponiendo que: 
\begin{equation*}
    g_1= - \frac{b_2}{R}(1+\pi_2) + t_1 + m_1,
\end{equation*}
\begin{equation*}
        g_2= b_2- \frac{m_1}{(1+\pi_2)}  + m_2 + t_2,
\end{equation*}
donde: 

$g_1=g_2=0$.

\item Encuentre: \textbf{demanda de dinero para ambos periodos y oferta de dinero.}
\item Resuelva el equilibrio monetario.
\end{enumerate}
