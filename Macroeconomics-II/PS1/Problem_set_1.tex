\section*{Ejercicios de optimización}

\begin{enumerate}
    \item \textbf{Utilidad (Cobb--Douglas).} \\
    Una persona tiene utilidad
    \[
        U(x,y)=x^{1/2}y^{1/2},
    \]
    precios \(p_x=1\), \(p_y=2\) e ingreso \(M=40\). \\
    Maximizar \(U(x,y)\) sujeto a la restricción presupuestaria
    \[
        x+2y=40.
    \]
    \textit{Hint:} \(\mathcal{L}=x^{1/2}y^{1/2}+\lambda(40-x-2y)\).

    \item \textbf{Producción (no lineal).} \\
    Función de producción
    \[
        f(x,y)=x^{0.3}y^{0.7}.
    \]
    Restricción de recursos:
    \[
        2x+3y=120.
    \]
    Encontrar \(x,y\) que maximicen \(f(x,y)\) sujeto a la restricción. \\
    \textit{Hint:} \(\mathcal{L}=x^{0.3}y^{0.7}+\lambda(120-2x-3y)\).

    \item \textbf{Utilidad (log).} \\
    Consumidor con utilidad
    \[
        U(x,y)=\ln x+\ln y,
    \]
    precios \(p_x=2\), \(p_y=1\) e ingreso \(M=60\). \\
    Maximizar \(U(x,y)\) sujeto a
    \[
        2x+y=60.
    \]
    \textit{Hint:} \(\mathcal{L}=\ln x+\ln y+\lambda(60-2x-y)\). (recuerda \(x,y>0\)).
\end{enumerate}

\item La empresa productora de bienes y servicios desea maximizar sus beneficios y está sujeto a su función de producción como se muestra a continuación.(considere dos periodos, donde t = 1, 2).

\end{equation*}
\begin{align*}
    s.a \hspace{1cm} Y_t=A_t K_t^\alpha N^{1-\alpha}_t
\end{align*}

\begin{align*}

\end{align*}
